# 1. Ansible — Create pathnex-admin

- name: Create Pathnex admin user
  hosts: all
  become: true

  vars:
    admin_user: pathnex-admin
    ssh_key: "ssh-rsa AAAA_REPLACE_WITH_PUBLIC_KEY"

  tasks:
    - name: Create user
      ansible.builtin.user:
        name: "{{ admin_user }}"
        shell: /bin/bash
        create_home: true

    - name: Add SSH key
      ansible.posix.authorized_key:
        user: "{{ admin_user }}"
        key: "{{ ssh_key }}"

  name: pathnex-app

  replicas: 2

      app: pathnex-app

# 5. Jenkins Pipeline
This combines the four tasks into a Declarative Pipeline.

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM          = "DevOps"
        ENV           = "prod"
        PROJECT       = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {
            steps {
                echo "Creating ${PROJECT} infrastructure"

                dir('terraform') {
                    sh 'terraform init'
                    sh 'terraform apply -auto-approve'
                }
            }
        }

        stage('App Deployment') {
            steps {
                echo "Deploying ${INSTITUTE_NAME} application"

                dir('ansible') {
                    sh 'ansible-playbook user.yml'
                }

                sh 'kubectl apply -f kubernetes/deployment.yml'
            }
        }

        stage('Monitoring Checks') {
            steps {
                sh '''
                    echo "===== Monitoring Menu ====="
                    echo "1) IP"
                    echo "2) Disk"
                    echo "3) Memory"

                    read -p "Enter choice: " choice

                    case "$choice" in
                      1) hostname -I ;;
                      2) df -h ;;
                      3) free -h ;;
                      *) echo "Invalid choice" ;;
                    esac
                '''
            }
        }
    }
}


# 6. GitLab CI
Create .gitlab-ci.yml:

variables:
  INSTITUTE_NAME: "Pathnex"
  TEAM: "DevOps"
  ENV: "prod"
  PROJECT: "Pathnex-Training"

stages:
  - infrastructure
  - app_deployment
  - monitoring_checks

infrastructure:
  stage: infrastructure
  script:
    - echo "Creating ${PROJECT} infrastructure"
    - cd terraform
    - terraform init
    - terraform apply -auto-approve

app_deployment:
  stage: app_deployment
  script:
    - echo "Deploying ${INSTITUTE_NAME} application"
    - cd ansible
    - ansible-playbook user.yml
    - cd ..
    - kubectl apply -f kubernetes/deployment.yml

monitoring_checks:
  stage: monitoring_checks
  script:
    - |
      echo "===== Monitoring Menu ====="
      echo "1) IP"
      echo "2) Disk"
      echo "3) Memory"

      read -p "Enter choice: " choice

      case "$choice" in
        1) hostname -I ;;
        2) df -h ;;
        3) free -h ;;
        *) echo "Invalid choice" ;;
      esac
